# 配置语音函数(MiniMax api)

```python

def text_to_speech(text):
    """
    调用 MiniMax TTS API 将文本转为语音（简化版）
    
    参数:
        text (str): 需要转换为语音的文本内容
        
    返回:
        bytes: MP3 音频数据的字节流 失败时返回 None
    """
    # 1. 从环境变量获取 MiniMax API Key（安全做法 避免硬编码）
    api_key = os.environ.get('MINIMAX_API_KEY')
    
    # 2. 若 API Key 不存在 显示错误并提前返回
    if not api_key:
        st.error("未找到 MINIMAX_API_KEY")
        return None

    # 3. MiniMax TTS 接口地址（注意域名必须是 minimax.chat）
    url = "https://api.minimax.chat/v1/t2a_v2"
    
    # 4. 请求头: Bearer Token 鉴权 + JSON 内容类型
    headers = {
        "Authorization": f"Bearer {api_key}",
        "Content-Type": "application/json"
    }
    
    # 5. 请求体参数（根据 MiniMax API 文档构建）
    payload = {
        "model": "speech-02-hd",                     # 使用高音质 TTS 模型
        "text": text,                                # 待合成的文本
        "stream": False,                             # 非流式 一次返回完整音频
        "voice_setting": {                           # 音色配置
            "voice_id": "female-shaonv",             # 少女音色 ID（可换成其他）
            "speed": 1.0,                            # 语速 范围 0.5~2.0
            "vol": 1.0,                              # 音量 范围 0~10
            "pitch": 0                               # 音调 范围 -12~12
        },
        "audio_setting": {                           # 音频格式配置
            "format": "mp3",                         # 输出 MP3 格式
            "sample_rate": 32000                     # 采样率 32kHz
        },
        "output_format": "hex"                       # 返回音频数据的编码格式:十六进制字符串
    }

    try:
        # 6. 发送 POST 请求 超时时间设为 30 秒
        resp = requests.post(url, json=payload, headers=headers, timeout=30)
        
        # 7. 如果 HTTP 状态码不是 2xx 则抛出异常（由 except 统一处理）
        resp.raise_for_status()
        
        # 8. 解析响应的 JSON 数据
        data = resp.json()
        
        # 9. 检查 MiniMax 业务层错误码（status_code == 0 表示成功）
        #    使用链式 get 避免 KeyError 默认 status_code 为 0（假设成功）
        if data.get("base_resp", {}).get("status_code", 0) != 0:
            # 提取错误信息并显示
            error_msg = data["base_resp"].get("status_msg", "未知错误")
            st.error(f"业务错误: {error_msg}")
            return None
        
        # 10. 从响应中取出十六进制编码的音频字符串 并转换为字节数据
        audio_bytes = bytes.fromhex(data["data"]["audio"])
        return audio_bytes
        
    except Exception as e:
        # 11. 捕获所有网络 解析等异常 显示通用错误信息
        st.error(f"语音合成失败: {e}")
        return None
```
