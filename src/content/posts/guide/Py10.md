## CSV文件操作

`.csv`（Comma-Separated Values，逗号分隔值）是一种简单、通用的文本文件格式，用于存储表格数据，可以直接使用Excel打开。

### CSV操作 - 方式一：文件操作的原始方式

#### 写入CSV

```python
# 写
with open("resources/01.csv", "w", encoding="utf-8") as f:
    f.write("姓名,年龄,性别,爱好\n")  # 写入表头
    f.write("小王,18,男,football\n")  # 写入数据
    f.write("小李,18,女,'Python,KPL'\n")  # 多个爱好用''包裹
    f.write("小张,18,男,C++\n")
    f.write("小王,20,男,HTML\n")
```

#### 读取CSV

```python
# 读
with open("resources/01.csv", "r", encoding="utf-8") as f:
    for line in f:
        print(line.strip())
```

### CSV操作 - 方式二：csv模块（推荐）

```python
# 导入Python内置的csv模块 用于读写CSV文件（不需要额外安装）
import csv

# 1. 写入CSV文件
# open() 打开文件
# "resources/02.csv" 文件路径和文件名
# "w" 以写入模式打开文件
# encoding="utf-8" 防止中文乱码
# newline="" 防止Windows系统下生成多余空行
with open("resources/02.csv", "w", encoding="utf-8", newline="") as f:
    # 创建DictWriter对象 专门用字典方式写入CSV
    # f 是文件对象
    # fieldnames 指定CSV表头（列名）
    writer = csv.DictWriter(f, fieldnames=["姓名", "年龄", "性别", "爱好"])

    # 写入表头（第一行：姓名,年龄,性别,爱好）
    writer.writeheader()

    # 写入一行数据 用字典格式 key必须和fieldnames对应
    writer.writerow({"姓名": "小王", "年龄": 18, "性别": "男", "爱好": "football"})
    writer.writerow({"姓名": "小王", "年龄": 20, "性别": "男", "爱好": "HTML"})
    writer.writerow({"姓名": "周翔", "年龄": 18, "性别": "男", "爱好": "C++"})
    writer.writerow({"姓名": "小王", "年龄": 18, "性别": "男", "爱好": "football,java"})

# 2.读取CSV文件
# "r" 以只读模式打开文件
with open("resources/02.csv", "r", encoding="utf-8") as f:
    # 创建DictReader对象 按字典方式读取CSV 自动把第一行当作key
    reader = csv.DictReader(f)

    # 循环读取每一行数据
    for row in reader:
        # 打印每一行 row是一个字典 可通过row["姓名"]获取对应值
        print(row)
```

## 获取电影榜单数据（基础版本）

```python
import requests
import csv
from lxml import html

# 常量 定义地址
MOVIE_LIST_FILE = "movie_list.csv"  # 存储电影列表文件
TMDB_TOP_URL_1 = "https://www.themoviedb.org/movie/top-rated"  # 默认访问第一页数据源
TMDB_TOP_URL_2 = "https://www.themoviedb.org/discover/movie/items"  # 第二页地址（开发者模式 network 下滑找items 有网址）
INDEX_URL = "https://www.themoviedb.org/"


# 获取电影详情数据 有返回值
def get_movie_info(movie_info_url):
    # 1.发送请求 获取电影详情数据
    print(f"发送请求{movie_info_url},获取电影详情数据")
    movie_response = requests.get(movie_info_url, timeout=60)
    # 2.解析数据 获取电影详情
    movie_doc = html.fromstring(movie_response.text)
    # 根据网页结构 随机应变
    # 电影名
    movie_names = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[1]/h2/a/text()')
    # 年份
    movie_years = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[1]/h2/span/text()')
    # 上映时间
    movie_times = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[1]/div/span[@class="release"]/text()')
    # 类型
    movie_tags = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[1]/div/span[@class="genres"]/a/text()')
    # 时长
    movie_cost_time = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[1]/div/span[@class="runtime"]/text()')
    # 评分
    movie_score = movie_doc.xpath('//*[@id="consensus_pill"]/div/div[1]/div/div/@data-percent')
    # 语言
    movie_languages = movie_doc.xpath('//section[@class="facts left_column"]/p[3]/text()')
    # 导演
    movie_directors = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[3]/ol/li[1]/p[1]/a/text()')
    # 小说作者
    movie_authors = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[3]/ol/li[2]/p[1]/a/text()')
    # 宣传语
    movie_slogans = movie_doc.xpath('//div[@class="header_info"]/h3[@class="tagline"]/text()')
    # 简介
    movie_outline = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[3]/div/p/text()')

    # 返回电影详情
    # 存放电影详情到字典
    movie_info = {
        "电影名": movie_names[0].strip() if movie_names else "",
        "年份": movie_years[0].strip() if movie_years else "",
        "上映时间": movie_times[0].strip() if movie_times else "",
        "类型": ",".join(movie_tags).strip() if movie_tags else "",
        "时长": movie_cost_time[0].strip() if movie_cost_time else "",
        "评分": movie_score[0].strip() if movie_score else "",
        "语言": movie_languages[0].strip() if movie_languages else "",
        "导演": ",".join(movie_directors).strip() if movie_directors else "",
        "小说作者": ",".join(movie_authors).strip() if movie_authors else "",
        "宣传语": movie_slogans[0].strip() if movie_slogans else "",
        "简介": movie_outline[0].strip() if movie_outline else ""
    }
    return movie_info


# 保存数据函数（csv文件）
def sava_all_movies(all_movies):
    with open(MOVIE_LIST_FILE, "w", encoding="utf-8", newline="") as csvfile:
        writer = csv.DictWriter(csvfile, fieldnames=["电影名", "年份", "上映时间", "类型", "时长", "评分", "语言", "导演", "小说作者", "宣传语", "简介"])
        writer.writeheader()  # 写入表头
        # 一次写入多行数据
        writer.writerows(all_movies)  # 写入数据


# 主函数 定义核心逻辑
def main():
    # 提高作用域 写在这里
    all_movies = []

    # 循环获取电影列表 1到5页
    # 浏览器开发者模式 network 中找下滑时新出现的items（有新地址） 然后在找payload中page（页）值不同（5页有100个电影） payload上面的view source点开（全部粘贴）复制到下面的data = ""中
    for page_num in range(1, 6):
        # 1.发送请求 获取电影榜单数据
        if page_num == 1:
            response = requests.get(TMDB_TOP_URL_1, timeout=60)
        else:
            # 换成post请求
            response = requests.post(TMDB_TOP_URL_2,
                                     # 修改page = f"{page_num}"请求不同页的电影榜单数据
                                     data=f"air_date.gte=&air_date.lte=&certification=&certification_country=CN&debug=&first_air_date.gte=&first_air_date.lte=&include_adult=false&include_softcore=false&latest_ceremony.gte=&latest_ceremony.lte=&page={page_num}&primary_release_date.gte=&primary_release_date.lte=&region=&release_date.gte=&release_date.lte=2026-12-07&show_me=everything&sort_by=vote_average.desc&vote_average.gte=0&vote_average.lte=10&vote_count.gte=300&watch_region=CN&with_genres=&with_keywords=&with_networks=&with_origin_country=&with_original_language=&with_watch_monetization_types=&with_watch_providers=&with_release_type=&with_runtime.gte=0&with_runtime.lte=400",
                                     timeout=60)
        print(f"发送请求,访问第{page_num}页数据,获取TMDB榜单数据")

        # 2.解析数据 把一大串HTML文本字符串 变成一棵可以用XPath查找的 网页树对象
        doc = html.fromstring(response.text)
        # 获取电影列表（一个一个div大盒子元素）
        movie_list = doc.xpath("//div[@class='media-list-results contents']/div")

        # 3.遍历电影列表 获取电影详情
        # .// 在当前 movie 里 不管嵌套多深 把所有 a 都找出来（子孙任意层）
        # ./ 只找当前 movie 的直接子节点 a（只能下一层）

        for movie in movie_list:
            movie_href = movie.xpath(".//a/@href")

            # 判断存在 后拼接成完整的电影详情地址
            if movie_href:
                movie_info_url = INDEX_URL + movie_href[0]
                # 发送请求 获取电影详情数据（定义个函数）
                movie_info = get_movie_info(movie_info_url)
                # 函数返回值添加到列表中（列表套字典）
                all_movies.append(movie_info)

    # 4.保存数据 存储在CSV文件中（上面的all_movies列表中的数据 平级）
    print("获取所有电影详情数据,保存到CSV文件中")
    sava_all_movies(all_movies)


# 测试
if __name__ == '__main__':
    main()
```

## 正则表达式

**正则表达式（Regular Expression）** 是一种由特定语法规则组成的文本模式，用来描述、匹配字符串中符合特定规则的字符序列。相当于一种模式匹配工具，允许用户通过简洁的语法进行复杂的文本搜索、匹配、提取和替换工作。

### 正则表达式示例

```
1[3-9]\d{9}
```

- `1`：第一位固定是数字 1
- `[3-9]`：第二位是 3~9 的数字
- `\d{9}`：后面紧跟 9 位任意数字

### re模块基本使用

```python
import re

s1 = "我的手机号是18809091231?我的另一个手机号18800001266,QQ号是1779989922你记住了吗"
s2 = "19163208515"

# match - 从字符串的开头开始匹配（返回Match对象）
# r是为了描述反斜杠真正的含义 也可以写\\
result1 = re.match("1[3-9]\\d{9}", s2)
print(result1.group())  # 获取匹配的结果
print(result1.span())   # 获取匹配项的索引（元组）
print(result1.start())  # 获取匹配项的索引的开始位置
print(result1.end())    # 获取匹配项的索引的结束位置

# search - 从任意位置开始 搜索第一个匹配项（返回Match对象）
result2 = re.search(r"1[3-9]\d{9}", s1)
print(result2.group())
print(result2.span())

# findall - 从任意位置开始 搜索所有匹配项（返回列表）
result3 = re.findall(r"1[3-9]\d{9}", s1)
print(result3)
```

### 正则表达式语法表

#### 字符匹配

| 表达式 | 描述 |
|--------|------|
| 普通字符 | 字母、数字、汉字，大多数字符，直接匹配自身 |
| `.` | 匹配任意一个字符（除`\n`） |
| `\d` | 匹配任意数字 0-9 |
| `\D` | 匹配非数字 |
| `\w` | 匹配单词字符，即a-z、A-Z、0-9、_ 和其他语言字符 |
| `\W` | 匹配非单词字符 |
| `[aeiou]` | 匹配其中的任何单个字符 |
| `[^aeiou]` | 求反，匹配不在字符列表中的任何单个字符 |
| `[0-5]` | 表示范围 |

#### 量词

| 表达式 | 描述 | 示例 |
|--------|------|------|
| `*` | 出现任意次数（0次或无数次） | `1[3-9]\d*` 以1开头，第二位是3~9，后面随便多少位数字（可以没有） |
| `+` | 至少出现一次（1次或无数次） | `1[3-9]\d+` |
| `?` | 至多出现一次（0次或一次） | `1[3-9]\d?` |
| `{m}` | 出现m次 | `1[3-9]\d{9}` |
| `{m,}` | 至少出现m次 | `1[3-9]\d{9,}` |
| `{m,n}` | 出现m到n次 | `1[3-9]\d{9,11}` |

#### 其他元字符

| 表达式 | 描述 | 示例 |
|--------|------|------|
| `\|` | 或的意思，匹配左右任意一个表达式 | `1(3\|5\|7\|8\|9)\d{9}` 第二位是3或5或7或8或9 |
| `()` | 分组，将括号里的多个字符视为一个单元 | `(ab+)` ab这个组合至少出现一次 |
| `^` | 匹配字符串开头 | `^1[3-9]\d{9}$` 必须以1开头，整体一共11位数字 |
| `$` | 匹配字符串结尾 | `^1[3-9]\d{9}$` 必须以1开头，整体一共11位数字 |

### 正则表达式练习

```python
import re

s1 = "18809090000是我的手机号,188开头的,以00结尾;我的另一个手机号是15500008888,两个QQ号分别是1259989092和13809091293821,邮箱为python666@163.com"

# 正则表达式
print(re.findall(r"188.*", s1))   # * 匹配任意个
print(re.findall(r"188.?", s1))   # ? 匹配0个或者1个（最多出现一次）
print(re.findall(r"188.+", s1))   # + 匹配1个或者多个（至少出现一次）

print(re.findall(r"188\d{8}", s1))     # {8} 匹配8个
print(re.findall(r"155\d{6,10}", s1))  # {8,10} 匹配8到10个
print(re.findall(r"155\d{6,}", s1))    # {6,} 匹配6个或者更多

print(re.findall(r"1[38]\d{8}", s1))     # [38] 匹配3或者8
print(re.findall(r"1[^38]\d{8}", s1))    # [^38] 匹配非3或者8
print(re.findall(r"1[3-9]\d{8}", s1))    # [3-9] 匹配3到9
print(re.findall(r"^1[3-9]\d{8}", s1))   # ^ 匹配开头 ['1880909000']
print(re.findall(r"1[3-9]\d{8}$", s1))   # $ 匹配结尾 [] 如果s1就是一串手机号可以匹配到

print(re.findall(r"\w+@\w+\.\w+", s1))  # \w+ 匹配任意字符（数字,字母,下划线和其他语言字符）\.表示转义 匹配['邮箱为python666@163.com']
print(re.findall(r"\w+@\w+\.\w+", s1, re.ASCII))  # 只匹配ASCII字符 不包含中文

# 注意
s2 = "现在的时间是2026-02-06 10:05:25,今天天气还可以,气温28度"
print(re.findall(r"\d{4}-\d{2}-\d{2}", s2))
# 分组（年）（月）（日）
# 正则里加了括号 () = 分组 → 只输出括号里的内容 不输出括号外的符号
print(re.findall(r"(\d{4})-(\d{2})-(\d{2})", s2))
```

## 获取电影榜单数据（正则表达式清洗数据）- 完整代码

```python
import re
import requests
import csv
from lxml import html

# 全局请求头
headers = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"
}

# 常量 定义地址
MOVIE_LIST_FILE = "movie_list.csv"  # 存储电影列表文件
TMDB_TOP_URL_1 = "https://www.themoviedb.org/movie/top-rated"  # 默认访问第一页数据源
TMDB_TOP_URL_2 = "https://www.themoviedb.org/discover/movie/items"  # 第二页地址（开发者模式 network 下滑找items 有网址）
INDEX_URL = "https://www.themoviedb.org/"


# 数据清洗 函数
# 清洗年份（去掉括号）
def get_movie_year(movie_years):
    movie_year = movie_years[0].strip() if movie_years else ""
    # 将括号替换为空串（去掉括号）
    return movie_year.replace("(", "").replace(")", "")


# 清洗上映时间（去掉US）
def get_movie_publish_date(movie_times):
    movie_time = movie_times[0].strip() if movie_times else ""
    # 正则表达式
    c = re.search(r"\d{4}-\d{2}-\d{2}", movie_time)
    n = c.group() if c else ""
    return n


# 清洗时长（全部换算为分钟）
def get_movie_cost_time(movie_cost_times):
    movie_cost_time = movie_cost_times[0].strip() if movie_cost_times else ""
    # 正则表达式
    # 提取小时
    h_res = re.search(r"(\d+)h", movie_cost_time)
    m_res = re.search(r"(\d+)m", movie_cost_time)
    # group(1)获取第一组结果 否则或获取h和m
    h = int(h_res.group(1)) if h_res else 0
    m = int(m_res.group(1)) if m_res else 0
    return h * 60 + m


# 获取电影详情数据 有返回值
def get_movie_info(movie_info_url):
    # 1.发送请求 获取电影详情数据
    print(f"发送请求{movie_info_url},获取电影详情数据")
    movie_response = requests.get(movie_info_url, timeout=60, headers=headers)
    # 2.解析数据 获取电影详情
    movie_doc = html.fromstring(movie_response.text)
    # 根据网页结构 随机应变
    # 电影名
    movie_names = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[1]/h2/a/text()')
    # 年份
    movie_years = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[1]/h2/span/text()')
    # 上映时间
    movie_times = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[1]/div/span[@class="release"]/text()')
    # 类型
    movie_tags = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[1]/div/span[@class="genres"]/a/text()')
    # 时长
    movie_cost_times = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[1]/div/span[@class="runtime"]/text()')
    # 评分
    movie_score = movie_doc.xpath('//*[@id="consensus_pill"]/div/div[1]/div/div/@data-percent')
    # 语言
    movie_languages = movie_doc.xpath('//section[@class="facts left_column"]/p[3]/text()')
    # 导演
    movie_directors = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[3]/ol/li[1]/p[1]/a/text()')
    # 小说作者
    movie_authors = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[3]/ol/li[2]/p[1]/a/text()')
    # 宣传语
    movie_slogans = movie_doc.xpath('//div[@class="header_info"]/h3[@class="tagline"]/text()')
    # 简介
    movie_outline = movie_doc.xpath('//*[@id="original_header"]/div[2]/section/div[3]/div/p/text()')

    # 数据清洗函数

    # 返回电影详情
    # 存放电影详情到字典
    movie_info = {
        "电影名": movie_names[0].strip() if movie_names else "",
        # 数据处理 单独写一个函数处理
        "年份": get_movie_year(movie_years),
        "上映时间": get_movie_publish_date(movie_times),
        "类型": ",".join(movie_tags).strip() if movie_tags else "",
        "时长": get_movie_cost_time(movie_cost_times),
        "评分": movie_score[0].strip() if movie_score else "",
        "语言": movie_languages[0].strip() if movie_languages else "",
        "导演": ",".join(movie_directors).strip() if movie_directors else "",
        "小说作者": ",".join(movie_authors).strip() if movie_authors else "",
        "宣传语": movie_slogans[0].strip() if movie_slogans else "",
        "简介": movie_outline[0].strip() if movie_outline else ""
    }
    return movie_info

        writer = csv.DictWriter(csvfile, fieldnames=["电影名", "年份", "上映时间", "类型", "时长", "评分", "语言", "导演", "小说作者", "宣传语", "简介"])

def main():
    # 提高作用域 写在这里
    all_movies = []

    # 循环获取电影列表 1到5页
    # 浏览器开发者模式 network 中找下滑时新出现的items（有新地址） 然后在找payload中page（页）值不同（5页有100个电影） payload上面的view source点开（全部粘贴）复制到下面的data = ""中
    for page_num in range(1, 6):
        # 1.发送请求 获取电影榜单数据
        if page_num == 1:
            response = requests.get(TMDB_TOP_URL_1, timeout=60, headers=headers)
        else:
            # 换成post请求
            response = requests.post(TMDB_TOP_URL_2,
                                     # 修改page = f"{page_num}"请求不同页的电影榜单数据
                                     data=f"air_date.gte=&air_date.lte=&certification=&certification_country=CN&debug=&first_air_date.gte=&first_air_date.lte=&include_adult=false&include_softcore=false&latest_ceremony.gte=&latest_ceremony.lte=&page={page_num}&primary_release_date.gte=&primary_release_date.lte=&region=&release_date.gte=&release_date.lte=2026-12-07&show_me=everything&sort_by=vote_average.desc&vote_average.gte=0&vote_average.lte=10&vote_count.gte=300&watch_region=CN&with_genres=&with_keywords=&with_networks=&with_origin_country=&with_original_language=&with_watch_monetization_types=&with_watch_providers=&with_release_type=&with_runtime.gte=0&with_runtime.lte=400",
                                     timeout=60, headers=headers)
        print(f"发送请求,访问第{page_num}页数据,获取TMDB榜单数据")

        # 2.解析数据 把一大串HTML文本字符串 变成一棵可以用XPath查找的 网页树对象
        doc = html.fromstring(response.text)
        # 获取电影列表（一个一个div大盒子元素）
        movie_list = doc.xpath("//div[@class='media-list-results contents']/div")

        # 3.遍历电影列表 获取电影详情
        # .// 在当前 movie 里 不管嵌套多深 把所有 a 都找出来（子孙任意层）
        # ./ 只找当前 movie 的直接子节点 a（只能下一层）

        for movie in movie_list:
            movie_href = movie.xpath(".//a/@href")

            # 判断存在 后拼接成完整的电影详情地址
            if movie_href:
                movie_info_url = INDEX_URL + movie_href[0]
                # 发送请求 获取电影详情数据（定义个函数）
                movie_info = get_movie_info(movie_info_url)
                # 函数返回值添加到列表中（列表套字典）
                all_movies.append(movie_info)

    # 4.保存数据 存储在CSV文件中（上面的all_movies列表中的数据 平级）
    print("获取所有电影详情数据,保存到CSV文件中")
    sava_all_movies(all_movies)
```
